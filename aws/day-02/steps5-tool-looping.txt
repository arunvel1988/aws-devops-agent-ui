from openai import OpenAI
import json

from agent.registry import TOOL_DEFINITIONS
from agent.executor import execute_tool

from memory.store import (
    get_response_id,
    save_response_id
)


client = OpenAI()


def run_agent(user_message, session_id="default"):

    # --------------------------------------------------
    # Load persistent conversation memory
    # --------------------------------------------------

    previous_response_id = get_response_id(session_id)

    print(f"Session: {session_id}")
    print(f"Previous response: {previous_response_id}")

    # --------------------------------------------------
    # First request
    # --------------------------------------------------

    if previous_response_id:

        response = client.responses.create(
            model="openai.gpt-oss-120b",
            previous_response_id=previous_response_id,
            input=[
                {
                    "role": "user",
                    "content": user_message
                }
            ],
            tools=TOOL_DEFINITIONS
        )

    else:

        response = client.responses.create(
            model="openai.gpt-oss-120b",
            input=[
                {
                    "role": "user",
                    "content": user_message
                }
            ],
            tools=TOOL_DEFINITIONS
        )

    # --------------------------------------------------
    # TOOL-CALLING LOOP
    # --------------------------------------------------

    while True:

        tool_outputs = []

        # ----------------------------------------------
        # Find tool calls
        # ----------------------------------------------

        for item in response.output:

            if item.type == "function_call":

                print(
                    f"Agent requested tool: {item.name}"
                )

                # Parse tool arguments
                arguments = json.loads(
                    item.arguments or "{}"
                )

                print(
                    f"Tool arguments: {arguments}"
                )

                # Execute tool
                result = execute_tool(
                    item.name,
                    arguments
                )

                print("AWS tool result:")
                print(result)

                # Add tool result
                tool_outputs.append(
                    {
                        "type": "function_call_output",
                        "call_id": item.call_id,
                        "output": json.dumps(result)
                    }
                )

        # ----------------------------------------------
        # No tool calls
        # Agent has finished
        # ----------------------------------------------

        if not tool_outputs:

            print("Agent finished tool execution.")

            save_response_id(
                session_id,
                response.id
            )

            return response.output_text

        # ----------------------------------------------
        # Send tool results back to model
        # ----------------------------------------------

        response = client.responses.create(
            model="openai.gpt-oss-120b",
            previous_response_id=response.id,
            input=tool_outputs,
            tools=TOOL_DEFINITIONS
        )
